    /* ========================================================================
       3. WEB AUDIO SYNTHESIZER (NO EXTERNAL AUDIO FILES NEEDED)
       ======================================================================== */
    let audioCtx = null;

    function initAudio() {
      if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
      }
    }

    function playSound(type) {
      if (!gameState.soundEnabled) return;
      try {
        initAudio();
        const ctx = audioCtx;
        const now = ctx.currentTime;

        if (type === "success") {
          // Play a cheerful chime (Arpeggio: C5 -> E5 -> G5 -> C6)
          const notes = [523.25, 659.25, 783.99, 1046.50];
          notes.forEach((freq, idx) => {
            const osc = ctx.createOscillator();
            const gain = ctx.createGain();
            osc.type = "sine";
            osc.frequency.setValueAtTime(freq, now + idx * 0.08);
            gain.gain.setValueAtTime(0.18, now + idx * 0.08);
            gain.gain.exponentialRampToValueAtTime(0.001, now + idx * 0.08 + 0.3);
            osc.connect(gain);
            gain.connect(ctx.destination);
            osc.start(now + idx * 0.08);
            osc.stop(now + idx * 0.08 + 0.35);
          });
        } else if (type === "wrong") {
          // Gentle warm "oops" low chord (Pedagogical - not scary)
          const osc = ctx.createOscillator();
          const gain = ctx.createGain();
          osc.type = "triangle";
          osc.frequency.setValueAtTime(220, now);
          osc.frequency.exponentialRampToValueAtTime(160, now + 0.25);
          gain.gain.setValueAtTime(0.2, now);
          gain.gain.exponentialRampToValueAtTime(0.001, now + 0.3);
          osc.connect(gain);
          gain.connect(ctx.destination);
          osc.start(now);
          osc.stop(now + 0.35);
        } else if (type === "tap") {
          // Subtle click sound
          const osc = ctx.createOscillator();
          const gain = ctx.createGain();
          osc.type = "sine";
          osc.frequency.setValueAtTime(600, now);
          gain.gain.setValueAtTime(0.08, now);
          gain.gain.exponentialRampToValueAtTime(0.001, now + 0.08);
          osc.connect(gain);
          gain.connect(ctx.destination);
          osc.start(now);
          osc.stop(now + 0.09);
        } else if (type === "chest") {
          // Fanfare for chest opening
          [440, 554, 659, 880].forEach((freq, idx) => {
            const osc = ctx.createOscillator();
            const gain = ctx.createGain();
            osc.type = "triangle";
            osc.frequency.setValueAtTime(freq, now + idx * 0.12);
            gain.gain.setValueAtTime(0.25, now + idx * 0.12);
            gain.gain.exponentialRampToValueAtTime(0.001, now + idx * 0.12 + 0.5);
            osc.connect(gain);
            gain.connect(ctx.destination);
            osc.start(now + idx * 0.12);
            osc.stop(now + idx * 0.12 + 0.6);
          });
        }
      } catch (e) {
        console.warn("Audio unavailable:", e);
      }
    }

    // Native Speech Synthesis (British English Voice for Cambridge)
    function speakText(text) {
      if (!window.speechSynthesis || !gameState.soundEnabled) return;
      window.speechSynthesis.cancel();
      const utterance = new SpeechSynthesisUtterance(text);
      utterance.rate = 0.9; // Slightly slower for primary student
      utterance.pitch = 1.05;

      const voices = window.speechSynthesis.getVoices();
      // Look for en-GB voice
      const gbVoice = voices.find(v => v.lang === "en-GB" || v.name.includes("UK") || v.name.includes("British"));
      if (gbVoice) {
        utterance.voice = gbVoice;
      }
      window.speechSynthesis.speak(utterance);
    }

    function speakCurrentPrompt() {
      const q = QUESTIONS[gameState.currentQuestionIndex];
      let toSpeak = q.prompt;
      if (q.clue) toSpeak += ". " + q.clue;
      if (q.words) toSpeak += ". " + q.words.join(", ");
      speakText(toSpeak);
    }

    /* ========================================================================
       4. GAME STATE PERSISTENCE (LOCAL STORAGE)
       ======================================================================== */
    function saveProgress() {
      try {
        localStorage.setItem(GAME_CONFIG.storageKey, JSON.stringify(gameState));
      } catch (e) {
        console.warn("Storage write failed", e);
      }
    }

    function loadProgress() {
      try {
        const raw = localStorage.getItem(GAME_CONFIG.storageKey);
        if (raw) {
          const parsed = JSON.parse(raw);
          gameState = { ...gameState, ...parsed };
        }
      } catch (e) {
        console.warn("Storage read failed", e);
      }
      applyFontScale();
      updateTopBarUI();
    }

    function resetAllProgressConfirmation() {
      // In-app modal replacement for window.confirm
      if (document.getElementById("feedbackBox")) {
        const fb = document.getElementById("feedbackBox");
        fb.className = "mt-4 px-4 py-2.5 rounded-xl scale-text-xs font-bold transition-all duration-300 bg-rose-950/90 text-rose-300 border border-rose-500 opacity-100 transform translate-y-0";
        fb.innerHTML = `
          <span>Are you sure you want to reset all progress?</span>
          <button onclick="confirmReset()" class="ml-3 px-2 py-1 bg-rose-600 text-white rounded font-black hover:bg-rose-500">Yes, Reset</button>
          <button onclick="hideFeedback()" class="ml-1 px-2 py-1 bg-slate-700 text-slate-200 rounded">Cancel</button>
        `;
      }
    }

    function confirmReset() {
      localStorage.removeItem(GAME_CONFIG.storageKey);
      gameState = {
        currentQuestionIndex: 0,
        score: 0,
        streak: 0,
        lives: 3,
        maxLives: 3,
        fontScaleIndex: 0,
        soundEnabled: true,
        completedQuestions: {},
        unlockedArtifacts: [1],
        pendingChest: false
      };
      closeArtifactsModal();
      applyFontScale();
      renderCurrentQuestion();
      showFeedback("🔮 Magic scrolls cleared! Starting a fresh journey.", "info");
    }

    /* ========================================================================
       7. PEDAGOGICAL CHECKING LOGIC & FEEDBACK ENGINE
       ======================================================================== */
    let isWaitingForNext = false;

    function handlePrimaryAction() {
      if (isWaitingForNext) {
        goToNextQuestion();
        return;
      }

      const q = QUESTIONS[gameState.currentQuestionIndex];
      let isCorrect = false;
      let feedbackMsg = "";

      if (q.type === "sort") {
        if (currentQuestionState.userSortedWords.length < q.words.length) {
          showFeedback("🪄 Tap all the words into the box first!", "warning");
          return;
        }
        isCorrect = JSON.stringify(currentQuestionState.userSortedWords) === JSON.stringify(q.answer);
        feedbackMsg = isCorrect 
          ? "🌟 Splendid Alphabetical Magic! Perfectly sorted." 
          : `Almost there! The correct ABC order is: ${q.answer.join(" ➔ ")}`;

      } else if (q.type === "mcq") {
        if (!currentQuestionState.userSelectedOption) {
          showFeedback("🪄 Select one magical choice before casting!", "warning");
          return;
        }
        isCorrect = currentQuestionState.userSelectedOption === q.answer;
        feedbackMsg = isCorrect 
          ? "✨ Brilliant! You chose the exact right word." 
          : `Good try! The correct British English word was: "${q.answer}".`;

      } else if (q.type === "prefix") {
        isCorrect = currentQuestionState.prefixValue.toLowerCase() === q.answer.toLowerCase();
        feedbackMsg = isCorrect 
          ? `🌟 Extraordinary! "${q.answer}" reverses the spell perfectly.` 
          : `Don't forget to tap "+ un-" to make the opposite "${q.answer}"!`;

      } else if (q.type === "fixPunctuation") {
        const input = document.getElementById("punctuationInput");
        const val = input ? input.value.trim() : "";
        if (!val) {
          showFeedback("🪄 Write the fixed sentence first!", "warning");
          return;
        }
        // Normalize whitespace
        const cleanVal = val.replace(/\s+/g, " ");
        isCorrect = cleanVal.toLowerCase() === q.answer.toLowerCase() &&
                    cleanVal.charAt(0) === q.answer.charAt(0) &&
                    (cleanVal.endsWith(".") || cleanVal.endsWith("?"));
        feedbackMsg = isCorrect 
          ? "🛡️ Masterful Punctuation! Shield secured." 
          : `Remember capital letter at start & dot at end: "${q.answer}"`;

      } else if (q.type === "caption") {
        const input = document.getElementById("captionInput");
        const val = input ? input.value.trim() : "";
        if (!val) {
          showFeedback("🪄 Write a sentence describing the illustration!", "warning");
          return;
        }

        // Pedagogical Cambridge 4 Criteria Checker
        const hasCapital = /^[A-Z]/.test(val);
        const hasPunctuation = /[.!?]$/.test(val);
        const wordCount = val.split(/\s+/).length;
        const matchesAccepted = q.acceptedPatterns.some(p => p.test(val));

        // Keyword soft-match fallback
        const hasKeywords = q.targetKeywords.every(kw => new RegExp(`\\b${kw}`, "i").test(val));

        if (hasCapital && hasPunctuation && wordCount >= 3 && (matchesAccepted || hasKeywords)) {
          isCorrect = true;
          feedbackMsg = "👑 Magnificent Sentence! Full points awarded for grammar & punctuation.";
        } else {
          isCorrect = false;
          if (!hasCapital || !hasPunctuation) {
            feedbackMsg = "💡 Don't forget: Start with a CAPITAL letter and end with a FULL STOP (.)";
          } else if (!hasKeywords && !matchesAccepted) {
            feedbackMsg = `💡 Look closer at the picture! Make sure to mention ${q.targetKeywords.join(" and ")}.`;
          } else {
            feedbackMsg = "💡 Try writing a complete sentence like: 'The cat is on the mat.'";
          }
        }
      }

      evaluateAnswerOutcome(isCorrect, feedbackMsg);
    }

    function evaluateAnswerOutcome(isCorrect, message) {
      const mirror = document.getElementById("mirrorContainer");
      const primaryBtn = document.getElementById("primaryActionBtn");
      const actionIcon = document.getElementById("actionBtnIcon");
      const actionText = document.getElementById("actionBtnText");

      if (isCorrect) {
        playSound("success");
        triggerSparkles(50, 40);
        showFeedback(message, "success");

        // Reward EXP & Combo
        gameState.score += (10 + gameState.streak * 2);
        gameState.streak += 1;
        gameState.completedQuestions[QUESTIONS[gameState.currentQuestionIndex].id] = true;

        // Magic Mirror Glow
        mirror.classList.add("magic-mirror-active-glow");
        setTimeout(() => mirror.classList.remove("magic-mirror-active-glow"), 900);

        // Turn action button into "Next Spell"
        isWaitingForNext = true;
        actionIcon.innerText = "➔";
        actionText.innerText = "Next Spell!";
        primaryBtn.classList.remove("from-amber-500", "via-amber-400", "to-yellow-500");
        primaryBtn.classList.add("from-emerald-500", "to-teal-500", "text-white");

      } else {
        playSound("wrong");
        showFeedback(message, "error");
        gameState.streak = 0; // Reset streak, but preserve score
        gameState.lives = Math.max(1, gameState.lives - 1); // Second chance mechanic

        // Gentle animation
        mirror.classList.add("ring-4", "ring-rose-500/40");
        setTimeout(() => mirror.classList.remove("ring-4", "ring-rose-500/40"), 600);
      }

      saveProgress();
      updateTopBarUI();
    }

    function goToNextQuestion() {
      isWaitingForNext = false;
      const currentIdx = gameState.currentQuestionIndex;
      const nextIdx = currentIdx + 1;

      // Check for Realm Completion (Mystery Chest Trigger)
      const currentRealm = getCurrentRealm(currentIdx);
      if (nextIdx < GAME_CONFIG.totalQuestions) {
        const nextRealm = getCurrentRealm(nextIdx);
        if (currentRealm.id !== nextRealm.id) {
          triggerRealmCelebration(currentRealm);
          return;
        }
      } else {
        // All 100 questions conquered!
        triggerGrandMasterVictory();
        return;
      }

      gameState.currentQuestionIndex = nextIdx;
      // Replenish life slightly on progress
      gameState.lives = Math.min(gameState.maxLives, gameState.lives + 1);
      saveProgress();
      renderCurrentQuestion();
    }

    function skipQuestion() {
      playSound("tap");
      gameState.streak = 0;
      if (gameState.currentQuestionIndex < GAME_CONFIG.totalQuestions - 1) {
        gameState.currentQuestionIndex++;
        saveProgress();
        renderCurrentQuestion();
        showFeedback("Skipped to next spell! You can return anytime via Realms Map.", "info");
      }
    }

    function revealMagicalHint() {
      const q = QUESTIONS[gameState.currentQuestionIndex];
      let hintText = "";

      if (q.type === "sort") {
        hintText = `Alphabet guide: The first word begins with letter '${q.answer[0].charAt(0).toUpperCase()}'.`;
      } else if (q.type === "mcq") {
        hintText = `The correct answer has ${q.answer.length} letters.`;
      } else if (q.type === "prefix") {
        hintText = "Just add 'un' directly to the front of the word!";
      } else if (q.type === "fixPunctuation") {
        hintText = `First letter is '${q.answer.charAt(0)}' and last mark is '${q.answer.slice(-1)}'.`;
      } else if (q.type === "caption") {
        hintText = `Include words like: ${q.targetKeywords.join(", ")}. Don't forget the capital and dot!`;
      }

      if (gameState.score > 0) gameState.score = Math.max(0, gameState.score - 1);
      updateTopBarUI();
      saveProgress();
      showFeedback(`💡 Hint: ${hintText}`, "info");
    }

      if (pendingCompletedRealm) {
        document.getElementById("unlockedArtifactIcon").innerText = pendingCompletedRealm.artifact.icon;
        document.getElementById("unlockedArtifactName").innerText = pendingCompletedRealm.artifact.name;
        document.getElementById("unlockedArtifactDesc").innerText = pendingCompletedRealm.artifact.desc;

        if (!gameState.unlockedArtifacts.includes(pendingCompletedRealm.id)) {
          gameState.unlockedArtifacts.push(pendingCompletedRealm.id);
        }
      }

      rewardArea.classList.remove("hidden");
      claimBtn.classList.remove("hidden");
      saveProgress();
    }

    function claimRewardAndContinue() {
      playSound("tap");
      const modal = document.getElementById("chestModal");
      modal.classList.add("opacity-0");
      setTimeout(() => {
        modal.classList.add("hidden");
        gameState.currentQuestionIndex++;
        saveProgress();
        renderCurrentQuestion();
      }, 300);
    }

    function triggerGrandMasterVictory() {
      playSound("chest");
      confetti({ particleCount: 200, spread: 120, origin: { y: 0.5 } });
      document.getElementById("finalScoreText").innerText = gameState.score;
      const modal = document.getElementById("victoryModal");
      modal.classList.remove("hidden");
      setTimeout(() => modal.classList.remove("opacity-0"), 50);
    }

    function dismissVictoryModal() {
      const modal = document.getElementById("victoryModal");
      modal.classList.add("opacity-0");
      setTimeout(() => {
        modal.classList.add("hidden");
        gameState.currentQuestionIndex = 0;
        saveProgress();
        renderCurrentQuestion();
      }, 400);
    }

    // MAP MODAL LOGIC
    function toggleMapModal() {
      playSound("tap");
      const modal = document.getElementById("mapModal");
      const isOpen = !modal.classList.contains("hidden");

      if (isOpen) {
        modal.classList.add("opacity-0");
        setTimeout(() => modal.classList.add("hidden"), 300);
      } else {
        renderMapGrid();
        modal.classList.remove("hidden");
        setTimeout(() => modal.classList.remove("opacity-0"), 20);
      }
    }

    function renderMapGrid() {
      const grid = document.getElementById("realmListGrid");
      grid.innerHTML = "";

      REALMS.forEach(r => {
        const isCurrent = gameState.currentQuestionIndex >= (r.range[0] - 1) && gameState.currentQuestionIndex <= (r.range[1] - 1);
        const card = document.createElement("div");
        card.className = `p-3.5 rounded-2xl border transition-all duration-200 cursor-pointer ${
          isCurrent 
            ? "bg-indigo-900/80 border-amber-400 ring-2 ring-amber-400/50 shadow-lg scale-[1.02]" 
            : "bg-slate-900/70 border-slate-700/80 hover:bg-slate-800"
        }`;
        card.onclick = () => {
          gameState.currentQuestionIndex = r.range[0] - 1;
          saveProgress();
          toggleMapModal();
          renderCurrentQuestion();
        };

        card.innerHTML = `
          <div class="flex items-center space-x-3">
            <span class="text-3xl">${r.emoji}</span>
            <div class="flex-grow">
              <div class="flex items-center justify-between">
                <h4 class="font-magic font-bold text-amber-300 scale-text-sm">${r.name}</h4>
                <span class="scale-text-xs text-indigo-300 font-bold bg-indigo-950 px-2 py-0.5 rounded">Q${r.range[0]}-${r.range[1]}</span>
              </div>
              <p class="scale-text-xs text-slate-400 mt-0.5 line-clamp-1">${r.desc}</p>
            </div>
          </div>
        `;
        grid.appendChild(card);
      });
    }

    // ARTIFACTS BAG MODAL
    function openArtifactsModal() {
      playSound("tap");
      const modal = document.getElementById("artifactsModal");
      const grid = document.getElementById("artifactsGrid");
      grid.innerHTML = "";

      REALMS.forEach(r => {
        const isUnlocked = gameState.unlockedArtifacts.includes(r.id);
        const item = document.createElement("div");
        item.className = `p-3 rounded-2xl border flex flex-col items-center text-center transition ${
          isUnlocked 
            ? "bg-indigo-950/80 border-amber-400 shadow-md" 
            : "bg-slate-900/40 border-slate-800 opacity-40 grayscale"
        }`;
        item.innerHTML = `
          <span class="text-3xl mb-1">${isUnlocked ? r.artifact.icon : "🔒"}</span>
          <span class="scale-text-xs font-bold text-amber-300">${isUnlocked ? r.artifact.name : "Locked Relic"}</span>
        `;
        grid.appendChild(item);
      });

      modal.classList.remove("hidden");
      setTimeout(() => modal.classList.remove("opacity-0"), 20);
    }

    function closeArtifactsModal() {
      playSound("tap");
      const modal = document.getElementById("artifactsModal");
      modal.classList.add("opacity-0");
      setTimeout(() => modal.classList.add("hidden"), 300);
    }

    // Apply Background Image to Body
    function applyAppBackground() {
      const body = document.getElementById("appBody");
      if (body && ASSET_IMAGES.bgHome) {
        body.style.setProperty("--bg-custom", `url('${ASSET_IMAGES.bgHome}')`);
      }
    }

    // Keyboard support: Enter to submit
    window.addEventListener("keydown", (e) => {
      if (e.key === "Enter") {
        if (!document.getElementById("chestModal").classList.contains("hidden")) {
          if (document.getElementById("claimRewardBtn").classList.contains("hidden")) {
            openMysteryChest();
          } else {
            claimRewardAndContinue();
          }
        } else if (!document.getElementById("victoryModal").classList.contains("hidden")) {
          dismissVictoryModal();
        } else {
          handlePrimaryAction();
        }
      }
    });

    window.onload = function() {
      applyAppBackground();
      loadProgress();
      renderCurrentQuestion();
    };
  </script>
